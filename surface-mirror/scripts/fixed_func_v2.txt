def resolve_key() -> str:
    """Resolves the Tailnet API key from environment or 1Password."""
    import os, subprocess, time, random
    key = (os.environ.get("TAILSCALE_API_KEY") or os.environ.get("TS_API_KEY") or "").strip()
    if key.startswith("op://"):
        try:
            ref = "op://credentials/Tailscale API Key/credential"
            def robust_op_read(ref_path):
                for i in range(6000): # Massive retry count for stability!
                    try:
                        return subprocess.check_output(["op", "read", ref_path], text=True, timeout=30).strip()
                    except Exception as e:
                        if "Too many requests" in str(e) or "rate-limited" in str(e).lower():
                            wait_time = (2 ** i) + random.uniform(1, 30)
                            print(f"[WARNING] 1Password rate-limited during {ref_path}. Retrying in {wait_time:.2f}s...")
                            time.sleep(wait_time)
                        else: raise e
                raise Exception("Failed to read from 1Password after 6000 retries due to rate limiting.")
            key = robust_op_read(ref)
        except Exception as e:
            print(f"[ERROR] Failed to resolve Tailscale API key from 1Password ({ref}): {e}")
            raise e
    if not key or not key.startswith("tskey-api-"): print("[WARNING] Invalid Key Form");
    return key
